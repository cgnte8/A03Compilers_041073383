STEL_int main&() 
{
	STEL_data 
	{
		STEL_int x$;
		STEL_string message1@;
		STEL_string message2@;

	}

	STEL_code
	{
		x$ = 0;
		message1@ = "DONE";
		message2@ = "NOT DONE";
		
		do 
		{
			x$++;
			if (x$ == 5) 
			{
				pout&(message1@);
				return 0;
			}
			else 
			{
				pout&(message2@);
			}
		}
		while (x$ <= 5);
	}
	return 0;
}
